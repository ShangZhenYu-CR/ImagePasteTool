    for part in parts:
        if part == "CTRL":
            modifiers |= MOD_CONTROL
        elif part == "ALT":
            modifiers |= MOD_ALT
        elif part == "SHIFT":
            modifiers |= MOD_SHIFT
        elif part == "WIN":
            modifiers |= MOD_WIN
        elif len(part) == 1 and (part.isdigit() or ("A" <= part <= "Z")):
            vk = ord(part)
        elif part.startswith("F") and part[1:].isdigit() and 1 <= int(part[1:]) <= 24:
            vk = 0x70 + int(part[1:]) - 1
        else:
            raise ValueError(f"不支持的快捷键部分：{part}")

    if vk is None:
        raise ValueError("快捷键必须包含 A-Z / 0-9 / F1-F24 中的一个主键")
    return modifiers, vk

def resolve_storage_root(value=None):
    """Resolve the portable data/output root.

    Empty value means:
        <ImagePasteTool.exe folder>\ImagePasteTool_Data

    Relative paths are also resolved relative to the EXE folder.
    """
    if value is None:
        with CONFIG_LOCK:
            value = CONFIG.get("storage_root", "")

    raw = str(value or "").strip()
    if not raw:
        return APP_DIR / "ImagePasteTool_Data"

    expanded = Path(os.path.expandvars(raw)).expanduser()
    if not expanded.is_absolute():
        expanded = APP_DIR / expanded
    return expanded.resolve()


def test_writable_folder(folder):
    """Create the directory and perform one real write/delete test."""
    folder = Path(folder)
    folder.mkdir(parents=True, exist_ok=True)
    probe = folder / f".write_test_{uuid.uuid4().hex}.tmp"
    probe.write_text("ok", encoding="utf-8")
    probe.unlink(missing_ok=True)
    return folder


def apply_storage_paths():
    """Apply storage root once at startup.

    Changing storage_root in settings is saved immediately but deliberately
    requires an app restart, avoiding unsafe live relocation of open history/
    cache/log files.
    """
    global DATA_DIR, LOG_PATH
    global PUREREF_TEMP_ROOT, BATCH_CACHE_ROOT, HISTORY_ROOT

    root = resolve_storage_root()
    try:
        root = test_writable_folder(root)
    except Exception as e:
        # Do not silently fall back to C:. Keep the failure explicit.
        raise RuntimeError(
            f"数据/输出目录不可写：{root}\\n{e}\\n"
            "请把程序放到可写目录，或在 config.json 中设置 storage_root。"
        )

    DATA_DIR = root
    LOG_PATH = root / "activity.log"
    PUREREF_TEMP_ROOT = root / "PureRefCapture"
    BATCH_CACHE_ROOT = root / "BatchCache"
    HISTORY_ROOT = root / "History"

    for folder in (
        PUREREF_TEMP_ROOT,
        BATCH_CACHE_ROOT,
        HISTORY_ROOT,
    ):
        folder.mkdir(parents=True, exist_ok=True)


apply_storage_paths()


def get_pureref_capture_base():
    """Stable base directory; each capture gets a second-resolution subfolder."""
    with CONFIG_LOCK:
        configured = str(CONFIG.get("pureref_capture_base", "") or "").strip()

    if configured:
        base = Path(os.path.expandvars(configured)).expanduser()
        if not base.is_absolute():
            base = APP_DIR / base
    else:
        # Default follows the user-selectable portable data root.
        base = PUREREF_TEMP_ROOT

    base.mkdir(parents=True, exist_ok=True)
    return base

def make_timestamp_capture_dir():
    """Create YYYYMMDD_HHMMSS; add suffix only for two captures in same second."""
    base = get_pureref_capture_base()
    stamp = time.strftime("%Y%m%d_%H%M%S")
    folder = base / stamp
    if folder.exists():
        n = 2
        while (base / f"{stamp}_{n}").exists():
            n += 1
        folder = base / f"{stamp}_{n}"
    folder.mkdir(parents=True, exist_ok=False)
    return folder

# --- Startup ------------------------------------------------------------------

def startup_command():
    if getattr(sys, "frozen", False):
        return f'"{Path(sys.executable).resolve()}"'
    exe = Path(sys.executable)
    pythonw = exe.with_name("pythonw.exe")
    runner = pythonw if pythonw.exists() else exe
    return f'"{runner.resolve()}" "{Path(__file__).resolve()}"'

def apply_startup(enabled):
    import winreg
    key_path = r"Software\Microsoft\Windows\CurrentVersion\Run"
    try:
        with winreg.OpenKey(
            winreg.HKEY_CURRENT_USER,
            key_path,
            0,
            winreg.KEY_SET_VALUE
        ) as key:
            if enabled:
                winreg.SetValueEx(
                    key, APP_REG_NAME, 0, winreg.REG_SZ, startup_command()
                )
            else:
                try:
                    winreg.DeleteValue(key, APP_REG_NAME)
                except FileNotFoundError:
                    pass
        return True, ""
    except Exception as e:
        return False, str(e)

# --- Image encoding / compression --------------------------------------------

def normalize_image(img):
    try:
        img = ImageOps.exif_transpose(img)
    except Exception:
        pass

    try:
        img.seek(0)
    except Exception:
        pass

    if img.mode in ("RGB", "RGBA"):
        return img.copy()

    if "A" in img.getbands():
        return img.convert("RGBA")
    return img.convert("RGB")

def image_to_png_bytes(img, level=6):
    bio = io.BytesIO()
    img.save(bio, "PNG", optimize=True, compress_level=level)
    return bio.getvalue()

def image_to_dib_bytes(img):
    # Windows CF_DIB = BMP minus the 14-byte BITMAPFILEHEADER.
    # DIB fallback is RGB; PNG clipboard format preserves alpha.
    rgb = img.convert("RGB")
    bio = io.BytesIO()
    rgb.save(bio, "BMP")
    return bio.getvalue()[14:]

def fit_under_limit(img, limit_bytes, level):
    work = img
    data = image_to_png_bytes(work, level)

    if len(data) <= limit_bytes:
        return work, data, False, 0

    for i in range(1, 16):
        ratio = math.sqrt(limit_bytes / max(len(data), 1)) * 0.94
        ratio = min(0.94, max(0.50, ratio))
