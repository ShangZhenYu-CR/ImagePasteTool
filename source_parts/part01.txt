def normalize_hotkey(text):
    text = str(text).upper().replace(" ", "")
    aliases = {"CONTROL": "CTRL", "WINDOWS": "WIN"}
    parts = [aliases.get(x, x) for x in text.split("+") if x]
    order = ["CTRL", "ALT", "SHIFT", "WIN"]
    mods = [x for x in order if x in parts]
    keys = [x for x in parts if x not in order]
    key = keys[-1] if keys else "V"
    return "+".join(mods + [key])
