                target=capture_pureref_selected_images,
                daemon=True
            ).start()

        elif msg.message == WM_HOTKEY and msg.wParam == ONE_BY_ONE_HOTKEY_ID:
            threading.Thread(
                target=convert_and_paste_one_by_one,
                daemon=True
            ).start()

        elif msg.message == WM_HOTKEY and msg.wParam == HISTORY_HOTKEY_ID:
            UI_QUEUE.put(("open_history", None))

        elif msg.message == WM_RELOAD_HOTKEY:
            user32.UnregisterHotKey(None, HOTKEY_ID)
            user32.UnregisterHotKey(None, PUREREF_HOTKEY_ID)
            user32.UnregisterHotKey(None, ONE_BY_ONE_HOTKEY_ID)
            user32.UnregisterHotKey(None, HISTORY_HOTKEY_ID)
            _registered_hotkey = None
            _registered_pureref_hotkey = None
            _registered_one_by_one_hotkey = None
            _registered_history_hotkey = None
            register_current_hotkeys()

    user32.UnregisterHotKey(None, HOTKEY_ID)
    user32.UnregisterHotKey(None, PUREREF_HOTKEY_ID)
    user32.UnregisterHotKey(None, ONE_BY_ONE_HOTKEY_ID)
    user32.UnregisterHotKey(None, HISTORY_HOTKEY_ID)


def request_hotkey_reload():
    if hotkey_thread_id:
        user32.PostThreadMessageW(
            hotkey_thread_id, WM_RELOAD_HOTKEY, 0, 0
        )

def create_tray_image():
    img = Image.new("RGBA", (64, 64), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    d.rounded_rectangle((9, 12, 55, 57), radius=9, fill=(64, 64, 64, 255))
    d.rounded_rectangle((20, 5, 44, 20), radius=5, fill=(120, 120, 120, 255))
    d.rectangle((16, 26, 48, 48), fill=(238, 238, 238, 255))
    d.polygon([(17, 47), (28, 35), (34, 41), (40, 35), (48, 47)],
              fill=(105, 105, 105, 255))
    d.ellipse((38, 29, 43, 34), fill=(105, 105, 105, 255))
    return img

def open_log(icon=None, item=None):
    try:
        if not LOG_PATH.exists():
            LOG_PATH.write_text("", encoding="utf-8")
        os.startfile(LOG_PATH)
    except Exception as e:
        notify(f"无法打开日志：{e}")

def tray_loop():
    global tray_icon
    menu = pystray.Menu(
        Item("设置…", open_settings_from_tray, default=True),
        Item(
            lambda item: f"历史剪贴板…  {CONFIG['history_hotkey']}",
            open_history_from_tray
        ),
        Item(lambda item: f"PureRef 捕获：{CONFIG['pureref_capture_hotkey']}",
             lambda icon, item: None, enabled=False),
        Item(lambda item: f"逐张下一张：{CONFIG['one_by_one_hotkey']}",
             lambda icon, item: None, enabled=False),
        pystray.Menu.SEPARATOR,
        Item("20 MB", tray_set_limit(20), checked=checked_limit(20), radio=True),
        Item("32 MB", tray_set_limit(32), checked=checked_limit(32), radio=True),
        Item("超限自动压缩", tray_toggle_compression, checked=checked_compression),
        Item("转换后自动粘贴", tray_toggle_auto_paste, checked=checked_auto_paste),
        pystray.Menu.SEPARATOR,
        Item("批量：一次性全粘贴", tray_set_batch_mode("all"),
             checked=checked_batch_mode("all"), radio=True),
        Item("批量：一次粘贴一张", tray_set_batch_mode("one"),
             checked=checked_batch_mode("one"), radio=True),
        Item(
            "批量：边处理边粘贴",
            tray_toggle_stream_processing,
            checked=checked_stream_processing
        ),
        pystray.Menu.SEPARATOR,
        Item("查看处理日志", open_log),
        Item("退出", exit_app)
    )
    tray_icon = pystray.Icon(
        APP_NAME, create_tray_image(), f"{APP_NAME} V{APP_VERSION}", menu
    )
    tray_icon.run()
